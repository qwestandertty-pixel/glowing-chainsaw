originEl.addEventListener("input", function() { clearTimeout(timeout); var q = (originEl.value || "").trim(); if (q.length < 2) { originSuggest.style.display = "none"; return; } timeout = setTimeout(function() { fetch("/api/locations/search?keyword=" + encodeURIComponent(q)) .then(function(r) { return r.json(); }) .then(function(data) { if (!Array.isArray(data) || data.length === 0) { originSuggest.style.display = "none"; return; } originSuggest.innerHTML = ""; data.forEach(function(item) { var btn = document.createElement("button"); btn.textContent = item.name + " (" + item.iataCode + ")"; btn.onclick = function() { originEl.value = item.iataCode; originSuggest.style.display = "none"; }; originSuggest.appendChild(btn); }); originSuggest.style.display = "block"; }) .catch(function(err) { console.error("origin suggest error", err); originSuggest.style.display = "none"; }); }, 300); });
